from sqlalchemy import (
    Column,
    Integer,
    String,
    DateTime,
    ForeignKey,
    Float,
    Boolean
)
from sqlalchemy.orm import relationship
from datetime import datetime, timezone

from app.database.db import Base


class User(Base):
    __tablename__ = "users"

    id = Column(Integer, primary_key=True, index=True)
    email = Column(String(255), unique=True, index=True, nullable=False)
    password_hash = Column(String(255), nullable=False)
    is_active = Column(Boolean, default=True, nullable=False)
    created_at = Column(
        DateTime,
        default=lambda: datetime.now(timezone.utc)
    )

    events = relationship("Event", back_populates="owner")


class Event(Base):
    __tablename__ = "events"

    id = Column(Integer, primary_key=True, index=True)
    event_name = Column(String, nullable=False)

    # Nullable temporarily to preserve existing events.
    owner_id = Column(
        Integer,
        ForeignKey("users.id"),
        nullable=True,
        index=True
    )

    created_at = Column(
        DateTime,
        default=lambda: datetime.now(timezone.utc)
    )

    owner = relationship("User", back_populates="events")
    photos = relationship("Photo", back_populates="event")


class Photo(Base):
    __tablename__ = "photos"

    id = Column(Integer, primary_key=True, index=True)
    filename = Column(String, nullable=False)
    file_path = Column(String, nullable=False)

    event_id = Column(Integer, ForeignKey("events.id"))

    event = relationship("Event", back_populates="photos")
    faces = relationship("Face", back_populates="photo")


class Face(Base):
    __tablename__ = "faces"

    id = Column(Integer, primary_key=True, index=True)
    photo_id = Column(Integer, ForeignKey("photos.id"))

    confidence = Column(Float)
    embedding_path = Column(String)

    photo = relationship("Photo", back_populates="faces")

def get_owned_event(
    db: Session,
    event_id: int,
    admin_id: int
):
    event = (
        db.query(Event)
        .filter(
            Event.id == event_id,
            Event.owner_id == admin_id
        )
        .first()
    )

    if event is None:
        raise HTTPException(
            status_code=404,
            detail="Event not found."
        )

    return event


def event_summary(db: Session, event: Event):

    photo_count = (
        db.query(func.count(Photo.id))
        .filter(Photo.event_id == event.id)
        .scalar()
    )

    face_count = (
        db.query(func.count(Face.id))
        .join(Photo, Face.photo_id == Photo.id)
        .filter(Photo.event_id == event.id)
        .scalar()
    )

    return {
        "id": event.id,
        "event_name": event.event_name,
        "created_at": str(event.created_at),
        "total_photos": photo_count,
        "total_faces": face_count
    }

@router.post("/")
def create_event(
    event_name: str,
    db: Session = Depends(get_db),
    current_admin: User = Depends(get_current_admin)
):

    event_name = event_name.strip()

    if not event_name:
        raise HTTPException(
            status_code=400,
            detail="Event name cannot be empty."
        )

    if len(event_name) > 150:
        raise HTTPException(
            status_code=400,
            detail="Event name cannot exceed 150 characters."
        )

    existing_event = (
        db.query(Event)
        .filter(
            Event.event_name == event_name,
            Event.owner_id == current_admin.id
        )
        .first()
    )

    if existing_event:
        raise HTTPException(
            status_code=409,
            detail="You already have an event with this name."
        )

    event = Event(
        event_name=event_name,
        owner_id=current_admin.id
    )

    try:
        db.add(event)
        db.commit()
        db.refresh(event)

        return {
            "status": "success",
            "message": "Event created successfully.",
            "event": event_summary(db, event)
        }

    except Exception:
        db.rollback()
        raise HTTPException(
            status_code=500,
            detail="Unable to create event."
        )

def classify_match(similarity: float) -> str:

    if similarity >= STRONG_MATCH_THRESHOLD:
        return "strong_match"

    if similarity >= POSSIBLE_MATCH_THRESHOLD:
        return "possible_match"

    return "low_match"

@router.get("/events/{event_id}/index-status")
def index_status(
    event_id: int,
    db: Session = Depends(get_db),
    current_admin: User = Depends(get_current_admin)
):

    get_owned_event(
        db,
        event_id,
        current_admin.id

    index_path = Path(
        f"indexes/event_{event_id}.faiss"
    )

    metadata_path = Path(
        f"indexes/event_{event_id}.json"
    )

    if not index_path.exists() or not metadata_path.exists():

        return {
            "event_id": event_id,
            "index_exists": False,
            "message": "Build the event index first."
        }

    try:

        with metadata_path.open(
            "r",
            encoding="utf-8"
        ) as file:

            metadata = json.load(file)

    except (json.JSONDecodeError, OSError):

        raise HTTPException(
            status_code=500,
            detail="Unable to read event index metadata."
        )

    return {
        "event_id": event_id,
        "index_exists": True,
        "total_embeddings": metadata.get(
            "total_embeddings",
            0
        )
    }

@router.post("/events/{event_id}/search")
async def search_photos(
    event_id: int,

    file: UploadFile = File(...),

    top_k: int = Query(
        default=10,
        ge=1,
        le=20
    ),

    min_similarity: float = Query(
        default=POSSIBLE_MATCH_THRESHOLD,
        ge=-1.0,
        le=1.0,
        description="Minimum cosine similarity required for a result."
    ),

    db: Session = Depends(get_db),

    current_admin: User = Depends(get_current_admin)
):

    # ------------------------------------------
    # VERIFY EVENT OWNERSHIP
    # ------------------------------------------

    get_owned_event(
        db,
        event_id,
        current_admin.id
    )

    # ------------------------------------------
    # VALIDATE FILE
    # ------------------------------------------

    extension = Path(
        file.filename or ""
    ).suffix.lower()

    if extension not in ALLOWED_EXTENSIONS:

        raise HTTPException(
            status_code=400,
            detail="Unsupported image format."
        )

    content = await file.read(
        MAX_FILE_SIZE + 1
    )

    if len(content) > MAX_FILE_SIZE:

        raise HTTPException(
            status_code=413,
            detail="Maximum upload size is 50 MB."
        )

    if not content:

        raise HTTPException(
            status_code=400,
            detail="Uploaded file is empty."
        )

    UPLOAD_DIR.mkdir(
        parents=True,
        exist_ok=True
    )

    temp_path = (
        UPLOAD_DIR /
        f"{uuid.uuid4().hex}{extension}"
    )

    try:

        temp_path.write_bytes(content)

        # ------------------------------------------
        # DETECT FACES
        # ------------------------------------------

        engine = get_face_engine()

        detection = engine.process_image(
            temp_path
        )

        total_faces = detection["total_faces"]

        if total_faces == 0:

            raise HTTPException(
                status_code=422,
                detail=(
                    "No face detected. "
                    "Please upload a photograph containing a visible face."
                )
            )

        # ------------------------------------------
        # LOAD EVENT INDEX
        # ------------------------------------------

        search_engine = PersistentFaceIndex()

        try:

            search_engine.load_event_index(
                event_id
            )

        except FileNotFoundError:

            raise HTTPException(
                status_code=404,
                detail="Event search index not found."
            )

        # ------------------------------------------
        # SEARCH DETECTED FACES
        # ------------------------------------------

        matched_photos = {}

        for uploaded_face in detection["faces"]:

            query_embedding = (
                uploaded_face["embedding"]
            )

            matches = search_engine.search(
                query_embedding,
                top_k=top_k
            )

            for match in matches:

                similarity = float(
                    match["similarity"]
                )

                # ----------------------------------
                # FILTER LOW SIMILARITY
                # ----------------------------------

                if similarity < min_similarity:
                    continue

                record = (
                    db.query(Face, Photo)
                    .join(
                        Photo,
                        Face.photo_id == Photo.id
                    )
                    .filter(
                        Face.id == match["face_id"],
                        Photo.event_id == event_id
                    )
                    .first()
                )

                if record is None:
                    continue

                face, photo = record

                photo_id = photo.id

                # ----------------------------------
                # REMOVE DUPLICATE PHOTO RESULTS
                # Keep strongest matching face
                # ----------------------------------

                if (
                    photo_id not in matched_photos
                    or similarity > matched_photos[photo_id]["similarity"]
                ):

                    matched_photos[photo_id] = {

                        "photo_id": photo.id,

                        "face_id": face.id,

                        "filename": photo.filename,

                        "similarity": similarity,

                        "match_quality": classify_match(
                            similarity
                        ),

                        "image_url": (
                            f"/api/events/{event_id}/photos/{photo.id}"
                        )

                    }

        # ------------------------------------------
        # SORT MATCHES
        # ------------------------------------------

        results = sorted(
            matched_photos.values(),
            key=lambda item: item["similarity"],
            reverse=True
        )

        results = results[:top_k]

        # ------------------------------------------
        # ROUND SIMILARITY VALUES
        # ------------------------------------------

        for result in results:

            result["similarity"] = round(
                result["similarity"],
                4
            )

        # ------------------------------------------
        # RESPONSE
        # ------------------------------------------

        return {

            "event_id": event_id,

            "faces_detected": total_faces,

            "minimum_similarity": min_similarity,

            "total_matches": len(results),

            "matches": results

        }

    finally:

        await file.close()

        if temp_path.exists():
            temp_path.unlink()

@router.get(
    "/events/{event_id}/photos/{photo_id}"
)
def get_event_photo(
    event_id: int,
    photo_id: int,
    db: Session = Depends(get_db),
    current_admin: User = Depends(get_current_admin)
):

    # ------------------------------------------
    # VERIFY EVENT OWNERSHIP
    # ------------------------------------------

    event = get_owned_event(
        db,
        event_id,
        current_admin.id
    )

    # ------------------------------------------
    # FIND PHOTO
    # ------------------------------------------

    photo = (
        db.query(Photo)
        .filter(
            Photo.id == photo_id,
            Photo.event_id == event_id
        )
        .first()
    )

    if photo is None:

        raise HTTPException(
            status_code=404,
            detail="Photo not found."
        )

    # ------------------------------------------
    # VALIDATE IMAGE PATH
    # ------------------------------------------

    image_path = Path(
        photo.file_path
    ).resolve()

    event_root = (
        Path("storage/events") /
        event.event_name
    ).resolve()

    if not image_path.is_relative_to(event_root):

        raise HTTPException(
            status_code=403,
            detail="Photo access denied."
        )

    if not image_path.is_file():

        raise HTTPException(
            status_code=404,
            detail="Image file not found."
        )

    # ------------------------------------------
    # DETECT MIME TYPE
    # ------------------------------------------

    media_type, _ = mimetypes.guess_type(
        str(image_path)
    )

    # ------------------------------------------
    # RETURN ORIGINAL PHOTO
    # ------------------------------------------

    return FileResponse(
        path=str(image_path),
        media_type=media_type or "application/octet-stream",
        filename=photo.filename,
        content_disposition_type="inline"
    )

class RegisterRequest(BaseModel):
    email: str = Field(min_length=5, max_length=255)
    password: str = Field(min_length=12, max_length=128)


@router.post("/register")
def register_first_admin(
    data: RegisterRequest,
    db: Session = Depends(get_db)
):
    email = data.email.strip().lower()

    if (
        "@" not in email
        or email.startswith("@")
        or email.endswith("@")
        or "." not in email.split("@")[-1]
    ):
        raise HTTPException(
            status_code=422,
            detail="Enter a valid email address."
        )

    # Only the first admin can register publicly.
    if db.query(User).count() > 0:
        raise HTTPException(
            status_code=403,
            detail=(
                "Admin registration is closed. "
                "An administrator account already exists."
            )
        )

    existing = (
        db.query(User)
        .filter(User.email == email)
        .first()
    )

    if existing:
        raise HTTPException(
            status_code=409,
            detail="Email address is already registered."
        )

    admin = User(
        email=email,
        password_hash=hash_password(data.password),
        is_active=True
    )

    try:
        db.add(admin)
        db.flush()

        # Claim only currently unassigned legacy events.
        db.query(Event).filter(
            Event.owner_id.is_(None)
        ).update(
            {Event.owner_id: admin.id},
            synchronize_session=False
        )

        db.commit()
        db.refresh(admin)

    except Exception:
        db.rollback()
        raise HTTPException(
            status_code=500,
            detail="Unable to create administrator account."
        )

    token = create_access_token(admin.id)

    return {
        "status": "success",
        "message": "First administrator created successfully.",
        "access_token": token,
        "token_type": "bearer",
        "admin": {
            "id": admin.id,
            "email": admin.email
        }
    }


@router.post("/login")
def login_admin(
    form_data: OAuth2PasswordRequestForm = Depends(),
    db: Session = Depends(get_db)
):
    email = form_data.username.strip().lower()

    admin = (
        db.query(User)
        .filter(User.email == email)
        .first()
    )

    # Use the same response for unknown email and wrong password.
    if (
        admin is None
        or not admin.is_active
        or not verify_password(
            form_data.password,
            admin.password_hash
        )
    ):
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Incorrect email or password.",
            headers={"WWW-Authenticate": "Bearer"}
        )

    token = create_access_token(admin.id)

    return {
        "access_token": token,
        "token_type": "bearer",
        "admin": {
            "id": admin.id,
            "email": admin.email
        }
    }

from datetime import datetime, timedelta, timezone

from argon2 import PasswordHasher
from argon2.exceptions import VerifyMismatchError, VerificationError
from jose import jwt

from app.core.config import (
    JWT_SECRET_KEY,
    JWT_ALGORITHM,
    ACCESS_TOKEN_EXPIRE_MINUTES
)


password_hasher = PasswordHasher()


def hash_password(password: str) -> str:
    return password_hasher.hash(password)


def verify_password(
    plain_password: str,
    hashed_password: str
) -> bool:
    try:
        return password_hasher.verify(
            hashed_password,
            plain_password
        )
    except (VerifyMismatchError, VerificationError):
        return False


def create_access_token(user_id: int) -> str:

    expires_at = (
        datetime.now(timezone.utc)
        + timedelta(minutes=ACCESS_TOKEN_EXPIRE_MINUTES)
    )

    payload = {
        "sub": str(user_id),
        "exp": expires_at
    }

    return jwt.encode(
        payload,
        JWT_SECRET_KEY,
        algorithm=JWT_ALGORITHM
    )


def decode_access_token(token: str):

    return jwt.decode(
        token,
        JWT_SECRET_KEY,
        algorithms=[JWT_ALGORITHM]
    )

def convert_file_schema_to_binary(schema):

    if isinstance(schema, dict):

        if schema.get("contentMediaType") == "application/octet-stream":
            schema.pop("contentMediaType", None)
            schema["format"] = "binary"

        for value in schema.values():
            convert_file_schema_to_binary(value)

    elif isinstance(schema, list):

        for item in schema:
            convert_file_schema_to_binary(item)


def custom_openapi():

    if app.openapi_schema:
        return app.openapi_schema

    openapi_schema = get_openapi(
        title=app.title,
        version=app.version,
        description=app.description,
        routes=app.routes,
        openapi_version="3.0.3"
    )

    convert_file_schema_to_binary(openapi_schema)

    app.openapi_schema = openapi_schema

    return app.openapi_schema


app.openapi = custom_openapi
